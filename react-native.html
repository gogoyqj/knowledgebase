3. C++ 层 — 纯 C++，无依赖

核心规则：.h 只放声明，.cpp 放实现。

CircleDrawer.h — 声明

#pragma once
#include <cstdint>
#include <vector>

namespace circleplugin {

struct CircleResult {
  std::vector<uint8_t> pixels;  // RGBA buffer
  int width;
  int height;
};

/**
 * @param radius  圆的半径（像素）
 * @param color   填充色，格式 0xRRGGBBAA
 * @return        RGBA 像素数据，尺寸 = 2*radius × 2*radius
 */
CircleResult drawCircle(int radius, uint32_t color);

}

CircleDrawer.cpp — 实现

#include "CircleDrawer.h"
#include <cmath>

namespace circleplugin {

CircleResult drawCircle(int radius, uint32_t color) {
  int size = radius * 2;
  CircleResult result;
  result.width = size;
  result.height = size;
  result.pixels.resize(size * size * 4, 0);  // RGBA, 4 bytes per pixel

  uint8_t r = (color >> 24) & 0xFF;
  uint8_t g = (color >> 16) & 0xFF;
  uint8_t b = (color >> 8)  & 0xFF;
  uint8_t a =  color        & 0xFF;

  float cx = radius - 0.5f;
  float cy = radius - 0.5f;
  float r2 = static_cast<float>(radius) * radius;

  for (int y = 0; y < size; y++) {
    for (int x = 0; x < size; x++) {
      float dx = x - cx;
      float dy = y - cy;
      if (dx * dx + dy * dy <= r2) {
        int idx = (y * size + x) * 4;
        result.pixels[idx + 0] = r;
        result.pixels[idx + 1] = g;
        result.pixels[idx + 2] = b;
        result.pixels[idx + 3] = a;
      }
    }
  }

  return result;
}

}

4. iOS 层 — ObjC++ 直调 C++

iOS 最简单，ObjC++ (.mm) 天然兼容 C++，直接 #include 就能调。

JS call → RCT_EXPORT_METHOD → C++ drawCircle() → Promise resolve 
// CirclePlugin.mm
#import "CirclePlugin.h"
#import <React/RCTLog.h>
#include "../cpp/CircleDrawer.h"  // 直接 include C++ 头文件

@implementation CirclePlugin

RCT_EXPORT_MODULE();

RCT_EXPORT_METHOD(drawCircle:(double)radius
                  color:(double)color
                  resolve:(RCTPromiseResolveBlock)resolve
                  reject:(RCTPromiseRejectBlock)reject)
{
  @try {
    auto result = circleplugin::drawCircle((int)radius, (uint32_t)color);

    NSData *data = [NSData dataWithBytes:result.pixels.data()
                                  length:result.pixels.size()];
    NSString *base64 = [data base64EncodedStringWithOptions:0];

    resolve(@{
      @"width": @(result.width),
      @"height": @(result.height),
      @"data": base64,
    });
  } @catch (NSException *exception) {
    reject(@"CIRCLE_ERROR", exception.reason, nil);
  }
}

@end
为什么 iOS 简单？ ObjC++ 文件（.mm）同时理解 Objective-C 和 C++ 语法，可以直接调用 C++ 函数、使用 C++ 类型。不需要额外的胶水层。

5. Android 层 — Java + JNI

Android 不能直接调 C++，必须经过 JNI（Java Native Interface）桥接。所以需要两层：

JS call → Java @ReactMethod → native method → JNI C++ 函数 → C++ drawCircle()

5.1 Java 模块 — CirclePluginModule.java

public class CirclePluginModule extends ReactContextBaseJavaModule {

    // 加载 CMake 编译出的 libcircleplugin.so
    static {
        System.loadLibrary("circleplugin");
    }

    // 声明 native 方法，C++ 侧实现
    private static native byte[] nativeDrawCircle(int radius, int color);

    @Override
    public String getName() {
        return "CirclePlugin";  // JS 侧模块名
    }

    @ReactMethod
    public void drawCircle(double radius, double color, Promise promise) {
        byte[] pixels = nativeDrawCircle((int)radius, (int)color);
        String base64 = Base64.encodeToString(pixels, Base64.NO_WRAP);
        // ... resolve promise
    }
}

5.2 JNI 桥接 — CirclePluginJNI.cpp

#include <jni.h>
#include "../../../../cpp/CircleDrawer.h"

extern "C" {

命名规则: Java_包名_类名_方法名

JNIEXPORT jbyteArray JNICALL
Java_com_circleplugin_CirclePluginModule_nativeDrawCircle(
    JNIEnv *env, jclass clazz, jint radius, jint color) {

    auto result = circleplugin::drawCircle(radius, (uint32_t)color);

    jbyteArray jPixels = env->NewByteArray(result.pixels.size());
    env->SetByteArrayRegion(jPixels, 0, result.pixels.size(),
                            (const jbyte *)result.pixels.data());
    return jPixels;
}

}

5.3 ReactPackage 注册 — CirclePluginPackage.java

public class CirclePluginPackage implements ReactPackage {
    @Override

        List<NativeModule> modules = new ArrayList<>();
        modules.add(new CirclePluginModule(ctx));
        return modules;
    }

    @Override

6. JNI 命名规则详解

JNI 靠函数名自动绑定 Java native 方法到 C++ 实现：

Java_包名_类名_方法名

包名中的 . → _

com.circleplugin.CirclePluginModule.nativeDrawCircle
     ↓             ↓                   ↓
com_circleplugin  CirclePluginModule  nativeDrawCircle

C++ 函数签名

Java_com_circleplugin_CirclePluginModule_nativeDrawCircle(
    JNIEnv *env,    // JNI 环境指针，操作 Java 对象用
    jclass clazz,   // 调用者类（static 方法用 jclass，实例方法用 jobject）
    jint radius,    // 对应 Java 的 int
    jint color      // 对应 Java 的 int
)

JNI 类型映射

JavaJNI C++ 类型签名字符

doublejdoubleD
booleanjbooleanZ
bytejbyteB
floatjfloatF
longjlongJ
shortjshortS
charjcharC
byte[]jbyteArray[B
int[]jintArray[I
StringjstringLjava/lang/String;
void—V

7. 替代方案：RegisterNatives

不想用命名规则？可以用 RegisterNatives 手动绑定，函数名随便取：

// C++ 函数名自由定义
jbyteArray drawCircleImpl(JNIEnv *env, jclass clazz, jint radius, jint color) {
    // 实现...
}

// JNI 类型签名: (参数类型)返回类型
static JNINativeMethod methods[] = {
    {"nativeDrawCircle", "(II)[B", (void *)drawCircleImpl}
};

// 在 JNI_OnLoad 中注册
jint JNI_OnLoad(JavaVM *vm, void *reserved) {
    JNIEnv *env;
    vm->GetEnv((void **)&env, JNI_VERSION_1_6);

    jclass clazz = env->FindClass("com/circleplugin/CirclePluginModule");
    env->RegisterNatives(clazz, methods, sizeof(methods) / sizeof(methods[0]));

    return JNI_VERSION_1_6;
}

JNI 签名 DSL 速查

(II)[B                    → (int, int) → byte[]
(D)V                      → (double) → void
(Ljava/lang/String;I)Z    → (String, int) → boolean
(IID)Ljava/lang/String;   → (int, int, double) → String
生成签名：不用手写，用 javap -s 自动生成：
javap -s com.circleplugin.CirclePluginModule

8. iOS vs Android 对比

iOSAndroid
桥接语言ObjC++ (.mm)Java + JNI
调 C++直接 #include通过 native 方法
胶水代码无JNI 函数 + Java 声明
编译Podspec 自动CMakeLists.txt
注册RCT_EXPORT_MODULEReactPackage
复杂度低高（多一层 JNI）
为什么 Android 需要 Java？ React Native Android 的模块系统是 Java 写的（ReactPackage、ReactModule），C++ 不是一等公民，必须通过 JNI 胶水层。iOS 的 ObjC++ 天然兼容 C++，不需要额外层。

9. 什么时候该用 C++？

场景建议
简单逻辑（画个圆、字符串处理）Java / Swift 直接写，不需要 C++
已有 C++ 库（OpenCV、FFmpeg）用 C++，避免重写
跨平台共享代码（iOS + Android + 桌面）用 C++，一份代码多端复用
性能密集型（音视频、物理模拟）用 C++，性能优势明显
想绕开 JNI 复杂度考虑 JSI / Expo Modules

10. 现代替代方案

JSI（JavaScript Interface）

绕过 Bridge，C++ 直接暴露给 JS，同步调用，零序列化：

JS → JSI → C++ HostObject → C++ 实现

Expo Modules

定义模块用 Swift/Kotlin，自动生成桥接代码：

// 不需要手写 JNI / ObjC++ 桥接
ExpoModule {
  Name("CirclePlugin")

  AsyncFunction("drawCircle") { radius: Int, color: Int ->
    // 直接写 Kotlin，Expo 自动生成 JS 桥接
  }
}

Nitro Modules

类似 JSI 但更高层，自动生成类型绑定：

// nitrogen 生成 C++ / Swift / Kotlin 绑定
interface CircleSpec extends HybridObject {
  drawCircle(radius: number, color: number): ArrayBuffer;
}
