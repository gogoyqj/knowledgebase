memory 内存问题与 OOM

OOM (Out of Memory) 是 Node.js 最常见的生产事故之一。V8 堆默认上限约 1.5GB（64位系统），超过即崩溃。

OOM 发生的三种场景

trending_up 场景一：堆内存缓慢泄漏

内存使用率缓慢上升不回落，几天后触发 OOM。最常见原因：闭包持有引用、缓存无限增长、事件监听器未移除。

flash_on 场景二：突发大内存分配

一次请求处理超大文件或数据集，瞬间超出堆限制。常见于 JSON.parse 大对象、文件读入内存。

warning 场景三：RSS 超限（非堆内存）

堆未满但 RSS（常驻内存）超出系统限制。通常由 Native Module（C++ addon）、Buffer 分配、SharedArrayBuffer 导致。Linux cgroup 限制下尤其常见。

运行时 CPU Profiling（不重启进程）

// 使用 inspector 模块在运行时采集 CPU Profile
const inspector = require('inspector');
const fs = require('fs');

async function startProfiling(durationMs = 10000) {
  const session = new inspector.Session();
  session.connect();
  session.post('Profiler.enable');

  session.post('Profiler.start');

  await new Promise(r => setTimeout(r, durationMs));

  const { profile } = await new Promise((resolve) => {
    session.post('Profiler.stop', (err, params) => resolve(params));
  });

  fs.writeFileSync(
    './cpu-profile.cpuprofile',
    JSON.stringify(profile)
  );
  session.disconnect();
  console.log('CPU profile saved.');
}

// 暴露为 HTTP 接口（生产环境安全起见加鉴权）
app.get('/debug/cpu-profile', async (req, res) => {
  const duration = parseInt(req.query.duration) || 10000;
  await startProfiling(duration);
  res.download('./cpu-profile.cpuprofile');
});

loop 事件循环阻塞

Node.js 单线程，事件循环被阻塞意味着所有请求都在排队等待。同步 I/O、密集计算、JSON.parse 大对象都是常见元凶。

检测事件循环阻塞

// 方式一：自定义事件循环延迟监控
let lastCheck = process.hrtime.bigint();

setInterval(() => {
  const now = process.hrtime.bigint();
  const delay = Number(now - lastCheck - 1000000000n) / 1e6;
  // delay 理论值 0ms（1s interval），实际值 > 50ms 说明阻塞
  if (delay > 50) {
    console.warn(`Event loop blocked for ${delay.toFixed(1)}ms`);
  }
  lastCheck = now;
}, 1000);

// 方式二：使用 blocked-at 模块（自动定位阻塞代码位置）
const blocked = require('blocked-at');
blocked((time, stack) => {
  console.warn(`Blocked for ${time}ms at:`, stack);
}, { threshold: 50 });

常见阻塞源与解决方案

阻塞源表现解决方案
JSON.parse() 大对象 处理大请求体时延迟飙升 流式 JSON 解析（如 JSONStream），限制请求体大小
fs.readFileSync() 磁盘 I/O 阻塞 改用 fs.promises.readFile()
正则回溯 (ReDoS) 特定输入导致正则卡死 使用 re2 库，避免嵌套量词
密集计算 CPU 密集型任务阻塞主线程 Worker Threads 或拆分为小任务用 setImmediate

Worker Threads 卸载 CPU 密集任务

// worker.js
const { parentPort } = require('worker_threads');

parentPort.on('message', (task) => {
  const result = heavyComputation(task.data);
  parentPort.postMessage(result);
});

// main.js — Worker 池模式
const { Worker } = require('worker_threads');
const os = require('os');

class WorkerPool {
  constructor(size = os.cpus().length) {
    this.workers = Array.from({ length: size }, () =>
      new Worker('./worker.js')
    );
    this.queue = [];
    this.idle = [...this.workers];
  }

  run(data) {
    return new Promise((resolve) => {
      const dispatch = () => {
        const worker = this.idle.pop();
        worker.once('message', (result) => {
          this.idle.push(worker);
          resolve(result);
          if (this.queue.length) this.queue.shift()();
        });
        worker.postMessage(data);
      };
      this.idle.length ? dispatch() : this.queue.push(dispatch);
    });
  }
}

Async Hooks 追踪 高级

// 追踪异步资源的创建和销毁
const async_hooks = require('async_hooks');

const active = new Map();
const hook = async_hooks.createHook({
  init(asyncId, type) {
    if (type === 'Timeout' || type === 'PROMISE') {
      active.set(asyncId, {
        type,
        stack: new Error().stack,
        created: Date.now(),
      });
    }
  },
  destroy(asyncId) {
    active.delete(asyncId);
  }
});
hook.enable();

// 定期检查未销毁的异步资源
setInterval(() => {
  if (active.size > 1000) {
    console.error(`Async leak: ${active.size} active resources`);
    // 输出最早创建的资源堆栈
    const oldest = [...active.values()][0];
    console.error(oldest.stack);
  }
}, 30000);

AbortController 超时控制

// 给异步操作加超时，防止永远挂起
async function fetchWithTimeout(url, timeoutMs = 5000) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), timeoutMs);

  try {
    const res = await fetch(url, { signal: controller.signal });
    return res;
  } finally {
    clearTimeout(timer);
  }
}

// 数据库查询超时

HTTP 连接泄漏排查

// 检查活跃的 socket 连接
const http = require('http');

// 自定义 Agent 追踪连接
const agent = new http.Agent({
  keepAlive: true,
  maxSockets: 50,
  maxFreeSockets: 10,
  timeout: 30000,
});

// 监控 socket 数量
setInterval(() => {
  const sockets = Object.entries(agent.sockets)
    .map(([k, v]) => `${k}: ${v.length}`);
  console.log('Active sockets:', sockets);
}, 10000);

// 常见泄漏原因：
// 1. 未调用 res.resume() 消费响应体 → socket 无法复用
// 2. 未正确处理错误事件 → socket 残留
// 3. keepAlive + 未设置 timeout → 连接不释放
// ✅ 安全的 HTTP 请求模板
async function safeRequest(url, options = {}) {
  const controller = new AbortController();
  const timer = setTimeout(() => controller.abort(), options.timeout || 10000);

  try {
    const res = await fetch(url, {
      ...options,
      signal: controller.signal,
    });

    // 必须消费响应体，否则 socket 不归还
    const body = await res.text();
    return { status: res.status, body };
  } catch (err) {
    if (err.name === 'AbortError') {
      throw new Error(`Request timeout: ${url}`);
    }
    throw err;
  } finally {
    clearTimeout(timer);
  }
}

优雅关闭 (Graceful Shutdown)

let isShuttingDown = false;

async function gracefulShutdown(signal) {
  if (isShuttingDown) return;
  isShuttingDown = true;
  console.log(`Received ${signal}. Shutting down gracefully...`);

  // 1. 停止接收新请求
  server.close(() => {
    console.log('HTTP server closed.');
  });

  // 2. 等待正在处理的请求完成（最多 30s）
  const deadline = setTimeout(() => {
    console.error('Shutdown timeout, forcing exit.');
    process.exit(1);
  }, 30000);

  try {
    // 3. 关闭数据库连接
    await pool.end();
    // 4. 关闭 Redis 连接
    await redis.quit();
    // 5. 刷新日志
    await logger.flush();

    console.log('Graceful shutdown complete.');
    clearTimeout(deadline);
    process.exit(0);
  } catch (err) {
    console.error('Error during shutdown:', err);
    process.exit(1);
  }
}

process.on('SIGTERM', () => gracefulShutdown('SIGTERM'));
process.on('SIGINT',  () => gracefulShutdown('SIGINT'));

clinic.js 诊断套件示例

# 安装
npm i -g clinic

# 1. 自动诊断 — 检测常见的性能问题
npx clinic doctor -- node app.js
# 自动生成报告，标识事件循环延迟、CPU 使用率等问题

# 2. 火焰图 — 定位 CPU 热点
npx clinic flame -- node app.js

# 3. 异步分析 — 可视化异步操作链
npx clinic bubbleprof -- node app.js

# 4. 对比两次运行结果
npx clinic flame --compare v1.flamegraph v2.flamegraph
checklist 问题排查清单

遇到问题时，按此清单逐步排查，快速缩小范围。

🔴 OOM 崩溃排查流程

search 确认 OOM 类型
→
memory Heap 还是 RSS?
→
camera 抓 Heap Snapshot
→
compare 对比分析
→
bug_report 定位泄漏源
  1. 查看进程日志中的 JavaScript heap out of memory 或 Allocation failed
  2. 检查 process.memoryUsage() 确认是 heap 还是 RSS 超限
  3. Heap 超限 → 用 v8.writeHeapSnapshot() 或 --heapprod 抓快照
  4. RSS 超限 → 检查 Buffer / Native Module / 共享内存分配
  5. Chrome DevTools 加载快照，Comparison 视图找增长最快的对象
  6. Retainers 视图追溯引用链，找到 GC Root 到泄漏对象的路径
  7. 修复后用 process.memoryUsage() 持续观察确认

🟡 响应变慢排查流程

timer 确认延迟来源
→
speed CPU 还是 I/O?
→
tune 对症下药
  1. 查看 APM 或日志中的请求延迟分位值（p50/p95/p99）
  2. CPU 高 → node --cpu-prof 或 0x 抓火焰图
  3. 事件循环延迟高 → monitorEventLoopDelay() 确认，blocked-at 定位阻塞源
  4. I/O 延迟高 → 检查数据库慢查询日志、外部 API 调用延迟
  5. GC 停顿长 → --trace-gc 分析 GC 频率和耗时，调整堆大小
  6. 连接池耗尽 → 监控 pool.waitingCount，调整 maxSockets / pool size

🟠 进程异常退出排查

  1. 检查退出码：0=正常，1=未捕获异常，128+signal=被信号杀死
  2. SIGKILL (137) → 通常是 OOM Killer，检查容器内存限制
  3. SIGSEGV (139) → 段错误，检查 Native Module 或 V8 Bug
  4. 查看 PM2 日志：pm2 logs --lines 200
  5. 生成 core dump：ulimit -c unlimited + --report-on-fatalerror
  6. 用 llnode 分析 core dump
