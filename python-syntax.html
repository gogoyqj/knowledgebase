scatter_plot

解构赋值

基础语法

Python 的解构比 JS 更强大，支持星号收集、嵌套解构。

javascript JavaScript
// 交换变量（ES6+）
[a, b] = [b, a];

// 没有原生 rest 解构到数组中间
const [first, ...rest] = [1,2,3,4];
// rest = [2,3,4] ✅
code Python
# 交换变量
a, b = b, a

# 星号收集 — 更灵活
first, *rest = [1, 2, 3, 4]
# rest = [2, 3, 4]

# 星号可以在任意位置
first, *mid, last = [1,2,3,4]
# mid = [2, 3]

# 嵌套解构
(a, b), c = [1, 2], 3
# a=1, b=2, c=3
filter_list

列表 / 字典 / 集合推导式

核心语法

用一行代码完成 filter + map，JS 需要链式调用多个方法。

javascript JavaScript
// 列表：filter + map
const result = arr
  .filter(x => x > 0)
  .map(x => x * 2);

// 对象：reduce 或 Object.fromEntries
const d = Object.fromEntries(
  items.filter(([, v]) => v > 0)
);
code Python
# 列表推导
result = [x * 2 for x in arr if x > 0]

# 字典推导
d = {k: v for k, v in items if v > 0}

# 集合推导
s = {x % 10 for x in arr}

# 嵌套推导
flat = [x for sub in matrix for x in sub]

# 生成器表达式（惰性求值）
total = sum(x * 2 for x in arr)
content_cut

切片 Slicing

核心语法

用 start:stop:step 语法操作序列，简洁到 JS 羡慕。

javascript JavaScript
// 取子数组
arr.slice(1, 3);

// 反转（会修改原数组）
arr.reverse();

// 隔一个取一个？自己写循环
arr.filter((_, i) => i % 2 === 0);
code Python
# 取索引 1-2
arr[1:3]

# 反转（不修改原数组）
arr[::-1]

# 步长 2，隔一个取一个
arr[::2]

# 取最后 3 个
arr[-3:]

# 浅拷贝
copy = arr[:]

# 字符串也能切
"hello"[1:4]  # "ell"
compare_arrows

三元表达式

语法差异

顺序完全不同：a if cond else b vs cond ? a : b

javascript JavaScript
const x = ok ? "yes" : "no";
code Python
x = "yes" if ok else "no"

⚠️ Python 的三元是一个表达式，不是语句。可以嵌套（但不建议）：

# 可以但别这么做
x = "a" if p else "b" if q else "c"
assignment

海象运算符 :=

3.8+

在表达式内部赋值并返回值，JS 没有对应语法。

javascript JavaScript
// 需要先声明再判断
const n = data.length;
if (n > 10) {
  console.log(`too long: ${n}`);
}

// 正则匹配？需要额外变量
const m = str.match(/\d+/);
if (m) {
  console.log(m[0]);
}
code Python
# 赋值同时判断
if (n := len(data)) > 10:
    print(f"too long: {n}")

# while 循环中读取
while (line := input()) != "quit":
    process(line)

# 列表推导中复用计算
results = [y for x in data
           if (y := transform(x)) > 0]

链式比较

语法差异

数学写法直接写进代码，JS 需要手动用 && 拼接。

javascript JavaScript
if (x > 0 && x < 10) { /* ... */ }

if (a <= b && b === c) { /* ... */ }
code Python
if 0 < x < 10:
    ...

# 可以链接多个
if a <= b == c:
    ...

# 等价于 a <= b and b == c
# 但 b 只求值一次！
call_split

多返回值 / 元组解包

核心语法

Python 函数天然支持返回多个值（本质是元组），直接解包。

javascript JavaScript
// 需要返回数组或对象
function divide(a, b) {
  return [Math.floor(a/b), a % b];
}
const [q, r] = divide(17, 5);
code Python
def divide(a, b):
    return a // b, a % b

q, r = divide(17, 5)
# q=3, r=2

# 丢弃不需要的值用 _
_, r = divide(17, 5)

# enumerate 也是多返回值
for i, val in enumerate(items):
    print(i, val)
auto_awesome

装饰器 @

核心语法

用 @ 语法给函数/类添加行为，等价于高阶函数包装。JS 需要手动调用或用实验性 decorators。

javascript JavaScript
// 手动包装
function logged(fn) {
  return (...args) => {
    console.log(`calling ${fn.name}`);
    return fn(...args);
  };
}
const home = logged(function home() {/*...*/});

// JS decorators（Stage 3，实验性）
@logged
function home() {/*...*/}
code Python
# 定义装饰器
def logged(fn):
    def wrapper(*args, **kwargs):
        print(f"calling {fn.__name__}")
        return fn(*args, **kwargs)
    return wrapper

@logged
def home(): ...

# 等价于 home = logged(home)

# 带参数的装饰器
@app.route("/")
def home(): ...

# 可叠加
@login_required
@cache(ttl=60)
def get_data(): ...
text_format

f-string 格式化

3.6+

比 JS 模板字符串更强大，支持表达式、格式化、调试输出。

javascript JavaScript
// 模板字符串
`hello ${name}`

// 格式化需要额外处理
price.toFixed(2)

// 调试输出？自己拼
console.log(`x=${x}, y=${y}`);
code Python
# 基础
f"hello {name}"

# 格式化小数位
f"price: {price:.2f}"

# 千分位分隔
f"count: {n:,}"

# 调试输出（3.8+）=号自动打印变量名
x, y = 1, 2
f"{x=}, {y=}"
# → "x=1, y=2"

# 表达式
f"result: {x + y:.2f}"
alt_route

match / case 模式匹配

3.10+

远超 JS 的 switch/case：支持解构、守卫条件、类型匹配。

javascript JavaScript
switch (command) {
  case "quit":
    exit();
    break;
  case "hello":
  case "hi":
    greet();
    break;
  default:
    // 守卫条件？不行
    // 解构？不行
}
code Python
match command:
    case "quit":
        exit()
    case "hello" | "hi":      # 多值
        greet()
    case str(s) if len(s) > 5: # 守卫
        print(s)

# 解构匹配
match point:
    case (0, 0):               # 字面量
        print("origin")
    case (x, 0) | (0, x):     # 变量绑定
        print(f"axis: {x}")
    case (x, y):
        print(f"{x}, {y}")

# 类匹配
match event:
    case Click(position=(x, y)):
        handle_click(x, y)
security

with 上下文管理器

核心语法

自动管理资源生命周期，替代 try/finally。JS 没有原生对应语法。

javascript JavaScript
// 没有 with，手动管理
let conn;
try {
  conn = await connect();
  await conn.query("...");
} finally {
  conn && conn.close();
}

// 或用 using（ES2024 提案）
using conn = await connect();
code Python
# with 自动管理
with open("file.txt") as f:
    data = f.read()
# 自动调用 f.close()，即使抛异常

# 多个资源
with open("in.txt") as fin, open("out.txt", "w") as fout:
    fout.write(fin.read())

# 异步版本
async with httpx.AsyncClient() as client:
    resp = await client.get("...")
sync

async / await 异步

语法相似

语法和 JS 几乎一样，但 Python 是协作式并发（单线程事件循环），需要显式 await。

javascript JavaScript
// 自动调度，无需显式让出
async function fetch() {
  const res = await fetch("/api");
  return res.json();
}

// 并发
const [a, b] = await Promise.all([
  fetch("/a"), fetch("/b")
]);
code Python
async def fetch():
    resp = await client.get("/api")
    return resp.json()

# 并发
a, b = await asyncio.gather(
    fetch("/a"), fetch("/b")
)

# 异步迭代
async for msg in websocket:
    await process(msg)

# 异步上下文
async with connect() as conn:
    await conn.query("...")
